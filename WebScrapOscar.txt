import requests
from bs4 import BeautifulSoup
import pandas as pd


# 1. Página principal
url = "https://www.scrapethissite.com/pages/ajax-javascript/"

resposta = requests.get(url)
resposta.raise_for_status()


# 2. Ler o HTML
soup = BeautifulSoup(resposta.text, "html.parser")


# 3. Encontrar os anos disponíveis
anos = []

for link in soup.find_all("a"):

    texto = link.get_text(strip=True)

    if texto.isdigit() and len(texto) == 4:
        anos.append(texto)


# Remove possíveis anos repetidos
anos = list(dict.fromkeys(anos))

print("Anos encontrados:", anos)


# 4. Lista onde vamos guardar todos os filmes
dados_filmes = []


# 5. Fazer a requisição AJAX para cada ano
for ano in anos:

    parametros = {
        "ajax": "true",
        "year": ano
    }

    resposta_ajax = requests.get(
        url,
        params=parametros
    )

    resposta_ajax.raise_for_status()

    filmes = resposta_ajax.json()


    # 6. Percorrer os filmes encontrados
    for filme in filmes:

        dados_filmes.append({
            "Ano": ano,
            "Título": filme["title"].strip(),
            "Indicações": filme["nominations"],
            "Prêmios": filme["awards"],
            "Melhor Filme": filme.get("best_picture", False)
        })


# 7. Criar o DataFrame
df = pd.DataFrame(dados_filmes)

print("\nTabela final:")
print(df)


# 8. Salvar em CSV
df.to_csv(
    "filmes_oscar.csv",
    index=False,
    encoding="utf-8-sig"
)
